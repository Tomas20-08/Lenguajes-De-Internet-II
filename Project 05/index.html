Formulario del Servidor Linux

Así está montado el servidor que recibe tus respuestas: Apache en CachyOS, un formulario en HTML, CSS y PHP, y MariaDB para guardar los datos.

# 1. Instalar Apache, PHP y MariaDB en CachyOS
sudo pacman -S apache php php-apache mariadb

# 2. Inicializar MariaDB
sudo mariadb-install-db --user=mysql --basedir=/usr --datadir=/var/lib/mysql

# 3. Activar Apache y MariaDB al iniciar el sistema
sudo systemctl enable --now httpd mariadb

# 4. Activar PHP en /etc/httpd/conf/httpd.conf
#    - comentar:  LoadModule mpm_event_module ...
#    - activar:   LoadModule mpm_prefork_module ...
LoadModule php_module modules/libphp.so
Include conf/extra/php_module.conf

# 5. Carpeta del formulario (raíz web: /srv/http)
sudo mkdir /srv/http/Server_003
sudo systemctl restart httpd

# 6. Probar desde la red local
http://172.22.113.166/Server_003/
-- Entrar a MariaDB desde la terminal:  sudo mariadb

CREATE DATABASE formulario_db CHARACTER SET utf8mb4;
USE formulario_db;

CREATE TABLE respuestas (
  id      INT AUTO_INCREMENT PRIMARY KEY,
  nombre  VARCHAR(100) NOT NULL,
  correo  VARCHAR(120) NOT NULL,
  mensaje TEXT,
  fecha   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Usuario que usa PHP (solo puede insertar y leer)
CREATE USER 'web'@'localhost' IDENTIFIED BY 'TU_CLAVE';
GRANT INSERT, SELECT ON formulario_db.* TO 'web'@'localhost';
FLUSH PRIVILEGES;

-- Ver lo que han enviado los usuarios
SELECT * FROM respuestas;
<?php
$ok = false;
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
  $db = new PDO('mysql:host=localhost;dbname=formulario_db;charset=utf8mb4', 'web', 'TU_CLAVE');
  $st = $db->prepare('INSERT INTO respuestas (nombre, correo, mensaje) VALUES (?, ?, ?)');
  $ok = $st->execute([$_POST['nombre'], $_POST['correo'], $_POST['mensaje']]);
}
?>
<!DOCTYPE html>
<html lang="es">
<head>
  <link rel="stylesheet" href="estilos.css">
</head>
<body>
  <form method="post">
    <input name="nombre" placeholder="Nombre" required>
    <input name="correo" type="email" required>
    <textarea name="mensaje"></textarea>
    <button>Enviar</button>
  </form>
  <?php if ($ok) echo '<p>Datos guardados</p>'; ?>
</body>
</html>
